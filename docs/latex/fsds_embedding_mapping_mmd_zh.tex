\begin{abstract}
\textbf{用什么神经网络表示不重要}（最后一层 hidden、层均值、外部 sentence embedding 均可），生产只需固定维度 $d$，并按逻辑段映射到 $Z$ 的每一行。
本文给出直观的 \textbf{ReAct 及其他 pattern 参数映射表}，以及代码中实现的 \textbf{MMD}$^2$、overlap ESS、episode 内 \textbf{shift/预算} 公式。
当 LIVE 窗在正确 segment 上注入漂移时，$X$ 批间 MMD 升高，且 $Z$ 上 shift 峰值与该段一致——与 demo 验收一致。
\end{abstract}

\section{两层契约（简述）}
\begin{itemize}[nosep]
  \item $Z\in\mathbb{R}^{B\times d}$：每段/角色/hop/节点一行，行 L2 归一化。
  \item $X_{\mathrm{ep}}$：\texttt{trace\_features} $=$ concat$(\mathrm{mean}(Z),$ 标量$)$；REF/LIVE 两批做监控。
  \item 辩论 early-stop：独立张量 $E\in\mathbb{R}^{R\times B\times d}$，\textbf{不能}代替预算用的 $Z$。
\end{itemize}

\section{Episode 内计算：shift 与预算}
行归一化 $\tilde{Z}_b = Z_b/\|Z_b\|_2$，质心 $c=\frac{1}{B}\sum_b \tilde{Z}_b$，
\begin{equation}
\label{eq:shift-zh}
s_b = \frac{\|\tilde{Z}_b - c\|_2}{\max_{b'} \|\tilde{Z}_{b'} - c\|_2 + \epsilon}.
\end{equation}
记 $\bar{s}_b = s_b/\sum_{b'} s_{b'}$，质量 $q_b$，关键度 $\eta_b$，
\begin{equation}
\label{eq:budget-zh}
L_b = \mathrm{clip}\big(\kappa\,\bar{s}_b\, q_b\, \eta_b,\; L_{\min,b},\; C_{\mathrm{lat}}\big).
\end{equation}
check 与 tier 随 $\bar{s}_b$ 与 $(1-q_b)$ 放大（\texttt{budget.py}）。

\section{批间监控：$X$ 上 MMD$^2$（REF vs LIVE）}
RBF 核 $k_\sigma(u,v)=\exp(-\|u-v\|^2/(2\sigma^2))$，实现 \texttt{rbf\_mmd2}（默认 $\sigma=1$）：
\begin{equation}
\label{eq:mmd-zh}
\widehat{\mathrm{MMD}}^2
= \frac{1}{n_0^2}\sum_{i,i'} k(X^0_i,X^0_{i'})
+ \frac{1}{n_1^2}\sum_{j,j'} k(X^1_j,X^1_{j'})
- \frac{2}{n_0 n_1}\sum_{i,j} k(X^0_i,X^1_j).
\end{equation}
\textbf{含义：}\eqref{eq:mmd-zh} 越大，说明 REF 窗与 LIVE 窗的 trace 分布差异越大（「MMD 升高」即指此量上升）。

同批数据上 RF 预测 $W\in\{\mathrm{REF},\mathrm{LIVE}\}$，得 domain AUC、列 VIMP；对 $\hat{e}(W|X)$ 做 overlap ESS，低于阈值则先 REFRESH 再 cap（\texttt{covariate\_attribution}）。

\section{注入漂移时的验收三联}
\begin{enumerate}
  \item \textbf{段级：}只扰动 $Z$ 的第 $b^\*$ 行 $\Rightarrow$ $\arg\max_b s_b = b^\*$（式 \eqref{eq:shift-zh}）。
  \item \textbf{批级：}LIVE 窗 episode 同样扰动 $\Rightarrow$ 式 \eqref{eq:mmd-zh} 相对 REF 升高。
  \item \textbf{策略级：}\texttt{fit\_agent\_plan} 相对 uniform 提高 $L_{b^\*}$/checks；写 \texttt{impact\_receipt}。
\end{enumerate}
