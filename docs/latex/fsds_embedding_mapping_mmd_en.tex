\begin{abstract}
The \emph{type} of neural representation (final hidden state, mean-pooled layer, or external sentence embedding) is intentionally unspecified: production must only keep dimension $d$ fixed and map each logical segment to one row of $Z$.
This note gives intuitive \textbf{parameter mapping tables} for ReAct and other agent patterns, plus the exact \textbf{MMD}$^2$, overlap ESS, and within-episode \textbf{shift/budget} formulas implemented in code.
When LIVE drift is injected on the correct segment, batch MMD$^2$ on $X$ rises and argmax shift score on $Z$ aligns with that segment---the acceptance test used in demos.
\end{abstract}

\section{Two-layer contract (recap)}
\begin{itemize}[nosep]
  \item $Z\in\mathbb{R}^{B\times d}$: one row per segment/role/hop/node; L2-normalized per row.
  \item $X_{\mathrm{ep}}\in\mathbb{R}^{p}$: \texttt{trace\_features} $=$ concat$(\mathrm{mean}(Z),\ \text{scalars})$; batches $X_{\mathrm{ref}}, X_{\mathrm{live}}$ for monitoring.
  \item Debate early-stop: separate tensor $E\in\mathbb{R}^{R\times B\times d}$ (not substituted for $Z$).
\end{itemize}

\section{Within-episode computation (shift and budget)}
Let $\tilde{Z}_b = Z_b/\|Z_b\|_2$ (code: \texttt{\_normalize\_rows}). Centroid $c=\frac{1}{B}\sum_b \tilde{Z}_b$,
\begin{equation}
\label{eq:shift}
s_b = \frac{\|\tilde{Z}_b - c\|_2}{\max_{b'} \|\tilde{Z}_{b'} - c\|_2 + \epsilon}.
\end{equation}
Normalized weights $\bar{s}_b = s_b/(\sum_{b'} s_{b'} + \epsilon)$. With quality $q_b\in(0,1]$, need $\eta_b$,
\begin{equation}
\label{eq:budget}
L_b = \mathrm{clip}\big(\kappa\,\bar{s}_b\, q_b\, \eta_b,\; L_{\min,b},\; C_{\mathrm{lat}}\big),
\end{equation}
check budget $\propto \bar{s}_b(1-q_b)$, tier from $(\bar{s}_b, q_b)$ thresholds (\texttt{budget.py}).

\section{Batch monitoring: MMD$^2$ on $X$ (REF vs LIVE)}
Stack $X_{\mathrm{ref}}\in\mathbb{R}^{n_0\times p}$, $X_{\mathrm{live}}\in\mathbb{R}^{n_1\times p}$. RBF kernel $k_\sigma(u,v)=\exp(-\|u-v\|^2/(2\sigma^2))$ (implementation: \texttt{rbf\_mmd2}, default $\sigma=1$):
\begin{equation}
\label{eq:mmd}
\widehat{\mathrm{MMD}}^2
= \frac{1}{n_0^2}\sum_{i,i'} k(X^0_i,X^0_{i'})
+ \frac{1}{n_1^2}\sum_{j,j'} k(X^1_j,X^1_{j'})
- \frac{2}{n_0 n_1}\sum_{i,j} k(X^0_i,X^1_j).
\end{equation}
\textbf{Interpretation:} larger \eqref{eq:mmd} $\Rightarrow$ covariate distribution of traces shifted between windows (``MMD rises'' in acceptance tests).

\paragraph{Domain classifier (same batch stack).}
$W_i\in\{0,1\}$ REF/LIVE labels; RF predicts $\hat{e}(W=1\mid X)$; report AUC and permutation VIMP on columns of $X$.
Overlap ESS on IPW weights $1/(\hat{e}(1-\hat{e}))$; gate if ESS fraction $<0.15$ (\texttt{covariate\_attribution}).

\paragraph{LOCO on columns of $X$.}
For feature column $j$, $\Delta_j = \widehat{\mathrm{MMD}}^2(X_{\mathrm{ref}}, X_{\mathrm{live}}) - \widehat{\mathrm{MMD}}^2(X_{\mathrm{ref}}^{(-j)}, X_{\mathrm{live}}^{(-j)})$; large $\Delta_j$ $\Rightarrow$ column $j$ of \texttt{trace\_features} carries shift (often aligns with drifted segment via $\mathrm{mean}(Z)$ block).

\section{Consistency check when drift is injected}
\begin{enumerate}
  \item \textbf{Segment test:} perturb only row $b^\*$ of $Z$ (e.g.\ observation or con) $\Rightarrow$ $b^\*=\arg\max_b s_b$ from \eqref{eq:shift}.
  \item \textbf{Batch test:} stack LIVE episodes with same perturbation in $Z$ $\Rightarrow$ $\widehat{\mathrm{MMD}}^2$ in \eqref{eq:mmd} increases vs REF (demos: Hillstrom-scale agent suites, legal block MMD${=}0.041$ vs finance $0.004$).
  \item \textbf{Policy test:} \texttt{fit\_agent\_plan} increases $L_{b^\*}$ or checks$_{b^\*}$ vs uniform plan; log \texttt{impact\_receipt}.
\end{enumerate}
